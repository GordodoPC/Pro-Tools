cmake_minimum_required(VERSION 3.22)
project(NovaAlphaSetup LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(NovaInstallSafety SHARED Safety.cpp)
target_compile_definitions(NovaInstallSafety PRIVATE UNICODE _UNICODE)
target_compile_options(NovaInstallSafety PRIVATE /utf-8)
target_link_libraries(NovaInstallSafety PRIVATE msi)
add_executable(NovaUninstall WIN32 Bootstrap.cpp version.rc)
target_compile_definitions(NovaUninstall PRIVATE NOVA_UNINSTALL=1 UNICODE _UNICODE)
target_compile_options(NovaUninstall PRIVATE /utf-8)
target_link_libraries(NovaUninstall PRIVATE shell32 ole32 msi)
set_target_properties(NovaUninstall PROPERTIES OUTPUT_NAME "Desinstalar NOVA DAW Alpha 0.7")
if(PAYLOAD_MSI)
    file(TO_CMAKE_PATH "${PAYLOAD_MSI}" PAYLOAD_RESOURCE)
    configure_file(payload.rc.in payload.rc @ONLY)
    add_executable(NovaSetup WIN32 Bootstrap.cpp version.rc "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
    target_compile_definitions(NovaSetup PRIVATE UNICODE _UNICODE)
    target_compile_options(NovaSetup PRIVATE /utf-8)
    target_link_libraries(NovaSetup PRIVATE shell32 ole32 msi)
    set_target_properties(NovaSetup PROPERTIES OUTPUT_NAME "NOVA-DAW-Alpha-0.7-Public-Test-x64-Setup")
endif()
